// Load user's completed bookings from Firestore
async function loadUserBookings(){
  const userData=JSON.parse(localStorage.getItem("tsyTaxiUser")||"{}");
  if(!userData.uid){
    bookingsList.innerHTML='<p style="text-align:center;padding:40px;color:#999">No bookings found</p>';
    return;
  }

  try{
    const{initializeApp}=await import("https://www.gstatic.com/firebasejs/10.14.1/firebase-app.js");
    const{getFirestore,collection,query,where,getDocs,orderBy}=await import("https://www.gstatic.com/firebasejs/10.14.1/firebase-firestore.js");

    const bookingsConfig={
      apiKey:"AIzaSyBwd7gY4QG96MKI-At38eXMVF-pQQU0F-s",
      authDomain:"tsytaxi-fcbd3.firebaseapp.com",
      projectId:"tsytaxi-fcbd3"
    };

    const bookingsApp=initializeApp(bookingsConfig,"bookingsModalApp");
    const db=getFirestore(bookingsApp);

    // Get completed trips
    const tripsRef=collection(db,"trips");
    const q=query(tripsRef,where("userId","==",userData.uid),orderBy("completedAt","desc"));
    const snapshot=await getDocs(q);

    if(snapshot.empty){
      bookingsList.innerHTML='<p style="text-align:center;padding:40px;color:#999">No completed trips yet</p>';
      document.getElementById("bookingCount").textContent="0";
      return;
    }

    document.getElementById("bookingCount").textContent=snapshot.size;

    bookingsList.innerHTML=snapshot.docs.map(doc=>{
      const trip=doc.data();
      const date=new Date(trip.completedAt).toLocaleDateString('en-IN',{day:'numeric',month:'short',year:'numeric'});
      return`
        <div style="background:#f9f9f9;border-radius:12px;padding:16px;margin-bottom:12px;border:2px solid #e9e9e9">
          <div style="display:flex;justify-content:space-between;margin-bottom:12px">
            <span style="font-size:12px;color:#777;font-weight:700">${date}</span>
            <span style="font-size:18px;font-weight:900;color:#16834a">₹${trip.fare.toFixed(2)}</span>
          </div>
          <div style="margin:8px 0">
            <div style="font-size:11px;color:#999;font-weight:700;margin-bottom:4px">FROM</div>
            <div style="font-size:13px;font-weight:700">${trip.pickup?.displayName||trip.pickup?.area||'N/A'}</div>
          </div>
          <div style="margin:8px 0">
            <div style="font-size:11px;color:#999;font-weight:700;margin-bottom:4px">TO</div>
            <div style="font-size:13px;font-weight:700">${trip.destination?.displayName||trip.destination?.area||'N/A'}</div>
          </div>
          <div style="display:flex;gap:20px;margin-top:12px;font-size:12px;color:#666">
            <span><strong>Distance:</strong> ${trip.distanceKm.toFixed(2)} km</span>
            <span><strong>Partner:</strong> ${trip.partnerName||'N/A'}</span>
          </div>
        </div>
      `;
    }).join('');

  }catch(error){
    console.error("Error loading bookings:",error);
    bookingsList.innerHTML='<p style="text-align:center;padding:40px;color:#e34c38">Error loading bookings</p>';
  }
}

// Review Modal Logic
let selectedRating=0;
let currentBookingId=bookingId;

document.querySelectorAll('.star').forEach(star=>{
  star.addEventListener('click',()=>{
    selectedRating=parseInt(star.dataset.rating);
    document.querySelectorAll('.star').forEach((s,idx)=>{
      s.style.opacity=idx<selectedRating?'1':'0.3';
    });
  });
});

document.getElementById('skipReview').addEventListener('click',()=>{
  document.getElementById('reviewModal').style.display='none';
});

document.getElementById('submitReview').addEventListener('click',async()=>{
  if(selectedRating===0){
    alert('Please select a rating');
    return;
  }

  const feedback=document.getElementById('reviewFeedback').value.trim();

  try{
    const{collection,addDoc,serverTimestamp}=await import("https://www.gstatic.com/firebasejs/10.14.1/firebase-firestore.js");

    await addDoc(collection(db,"reviews"),{
      bookingId:currentBookingId,
      userId:window.currentUser?.uid||'',
      rating:selectedRating,
      feedback:feedback,
      createdAt:serverTimestamp()
    });

    alert('Thank you for your feedback!');
    document.getElementById('reviewModal').style.display='none';
  }catch(error){
    console.error("Review error:",error);
    alert('Error submitting review');
  }
});

// Add to firestore.rules

match /reviews/{reviewId} {
  allow create: if request.resource.data.userId is string
    && request.resource.data.rating is number
    && request.resource.data.rating >= 1
    && request.resource.data.rating <= 5;
  allow read: if true;
}

match /coupons/{couponId} {
  allow read: if true;
  allow write: if true;  // Admin only in production
}

match /couponUsage/{usageId} {
  allow create: if request.resource.data.userId is string;
  allow read: if true;
}
